function Get-ArtifactSnapshot {
  param([string[]]$Roots)
  foreach ($root in $Roots) {
    $mem = Join-Path $root 'memory\governance'
    if (-not (Test-Path $mem)) { throw "memory/governance 不存在: $mem" }
    Get-ChildItem $mem -Recurse -File | ForEach-Object {
      [pscustomobject]@{
        root = $root
        relativePath = $_.FullName.Substring($mem.Length).TrimStart('\')
        sha256 = (Get-FileHash $_.FullName -Algorithm SHA256).Hash
        length = $_.Length
        lastWriteTimeUtc = $_.LastWriteTimeUtc.ToString('o')
      }
    }
  }
}
function Assert-ArtifactsUnchanged {
  param($Before, $After, [string]$Label)
  $b = $Before | Sort-Object root,relativePath | ConvertTo-Json -Depth 5 -Compress
  $a = $After  | Sort-Object root,relativePath | ConvertTo-Json -Depth 5 -Compress
  if ($b -ne $a) { throw "$Label: memory/audit 内容或 mtime 变化，卸载未做到只读保留" }
}

## 8. 风险与处置

| 风险 | 处置 |
|---|---|
| 误碰 desktop | 所有命令带 `--profile p4-*`；desktop 只读快照；发现 hash/计数变化立即停手 |
| `Remove-Item` 删到 junction 目标 | 只使用 `cmd /c rmdir` 删 junction；删后断言目标源码仍在 |
| pnpm/prune/redirect 漂移 | 不跑 pnpm/`dsh plugin add`；只用 file:+junction+bundle |
| v2 状态被 v0.2.1 当损坏 | 回滚前复制 v2 证据，恢复 `.v1.bak`；不在旧版上尝试迁移多根；只称兼容回滚 |
| 多根 UI/无 cwd | 每个 workspace 用独立 `.git`；若 `/cg roots` 只见一根，停手记录，不伪造 |
| configRoot 强制合并被误判 | nested `.git` 场景只允许 outer/outside 两根；nested 出现独立 root 即失败 |
| `/compact` 不触发 | 记 `not-triggered` 并单列“未验证”；不宣称四件套；引用 P3 测试但不冒充真机 |
| EBUSY/句柄 | 停实例后等待 2s，重试一次并记录；演练后连续 3 次全量 + teardown 复核；P4 观察项，不掩盖 |
| 只读根披露缺失 | 若 `protection=unavailable` 未出现/显示 `ok`，P4 gate 直接失败，停手报 00；先提窄修复，不改口径 |
| BOM/JSON | 所有 JSON 用 Node/UTF8NoBOM；装前装后都扫 BOM |
| 端口/进程冲突 | 固定隔离端口先检查；只杀本轮 PID；不碰 desktop 进程 |
| profile/state 残留 | 唯一名字；失败保留现场并报 00；不删除用户桌面文件 |
